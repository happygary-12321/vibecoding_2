def run(name, args):
    command = [sys.executable, *args]
    result = subprocess.run(command, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    (out / (name + ".command.txt")).write_text(subprocess.list2cmdline(command) + "\n", encoding="utf-8")
    (out / (name + ".log")).write_bytes(result.stdout)
    (out / (name + ".exit.txt")).write_text(str(result.returncode) + "\n")
    print(name, "exit", result.returncode, result.stdout.decode("utf-8", errors="replace")[-1200:], flush=True)
    return result

class StripDocstrings(ast.NodeTransformer):
    def generic_visit(self, node):
        super().generic_visit(node)
        if isinstance(node, (ast.Module, ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
            if node.body and isinstance(node.body[0],ast.Expr) and isinstance(node.body[0].value,ast.Constant) and isinstance(node.body[0].value.value,str):
                node.body.pop(0)
        return node

for f in files:
    old=subprocess.check_output(["git","show",baseline+":"+f]).decode("utf-8")
    new=pathlib.Path(f).read_text(encoding="utf-8")

    ast_results.append({"file":f,"executable_ast_equal":equal})
